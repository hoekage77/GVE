Data Models

Complete reference for all data structures

Session

A Session represents a user's conversation and generation context. It maintains message history, scene versions, and undo/redo state. 

interface SessionState {
  // Identifiers
  sessionId: string;              // UUID v4
  
  // Timestamps
  createdAt: ISOString;
  updatedAt: ISOString;
  
  // Status
  status: "idle" | "parsing" | "selecting" | "generating" | "executing";
  
  // Current Scene
  currentScene: SceneVersion | null;
  
  // Versioning System
  artifacts: Artifact[];        // Major versions collection
  artifactPointer: number;        // Current artifact index
  
  // Legacy flat versions (backward compat)
  versions: SceneVersion[];
  versionPointer: number;
  revisionPointer: number;
  
  // Chat History
  messages: SessionMessage[];
  
  // Debugging
  orchestrationTrace: TraceEntry[];
  
  // Metadata
  metadata?: Record;
}

Session Status States

Status Description
idle No active generation, waiting for input
parsing Intent parsing in progress
selecting Skill selection in progress
generating Code generation in progress
executing Sandbox execution in progress

Scene Version

A SceneVersion represents a snapshot of generated code with metadata about its origin and relationships to other versions. 

interface SceneVersion {
  // Identifiers
  id: string;                     // Version UUID
  
  // Versioning
  artifactIndex: number;          // Which artifact this belongs to
  revisionIndex: number;        // Position within artifact
  
  // Scene Content
  code: string;                   // Generated JavaScript code
  skill: string;                  // Skill ID (threejs, p5js, etc.)
  preview?: string;               // Preview URL or data
  
  // Context
  explanation?: string;           // Natural language explanation
  query?: string;                 // Original user query
  
  // Provenance
  source: "generation" | "edit" | "undo" | "redo" | "navigation";
  parentVersionId?: string;       // Previous version (for edits)
  
  // Timestamps
  createdAt: ISOString;
  
  // Execution metadata
  execution?: {
    sandboxId: string;
    duration: number;
    renderCount: number;
    status: "completed" | "error" | "timeout";
  };
}

Versioning Hierarchy

Artifact 0: "Solar System"
├── Revision 0: "Create solar system" (generation)
├── Revision 1: "Make sun bigger" (edit)
├── Revision 2: "Add Saturn rings" (edit)
└── [Current Position: Rev 2]

Artifact 1: "Animated Solar System"
├── Revision 0: "Animate the planets" (generation)
├── Revision 1: "Slow down orbits" (edit)
└── [Alternate branch]

Navigation:
- Undo/Redo: Within current artifact (Rev 2 → Rev 1 → Rev 0)
- Artifact Nav: Switch between artifacts (Art 1 ↔ Art 0)
- Global Nav: Flat version list across all artifacts

Artifact

An Artifact is a collection of related scene revisions, representing a major version or branch in the creative process. 

interface Artifact {
  id: string;                     // Artifact UUID
  name?: string;                  // Optional name/description
  
  // Revisions
  revisions: SceneVersion[];      // Ordered list of versions
  
  // Metadata
  skill: string;                  // Primary skill used
  createdAt: ISOString;
  
  // Statistics
  stats?: {
    totalRevisions: number;
    lastModified: ISOString;
  };
}

Session Message

Messages represent the chat history between user and assistant, including system messages, thoughts, and generation results. 

interface SessionMessage {
  // Identifiers
  id: string;                     // Message UUID
  correlationId?: string;         // Groups related messages
  
  // Content
  role: "user" | "assistant" | "system" | "thought";
  content: string;
  
  // Type
  kind?: "chat" | "generation" | "modification" | "explanation" | 
         "error" | "thought" | "system";
  
  // Scene reference (for generation messages)
  sceneId?: string;
  
  // Metadata
  metadata?: {
    skill?: string;
    versionId?: string;
    query?: string;
    [key: string]: any;
  };
  
  // Error info
  error?: SessionMessageError;
  
  // Timestamps
  createdAt: ISOString;
  updatedAt?: ISOString;
}

interface SessionMessageError {
  code: string;                   // Error code
  title: string;                    // Short title
  message: string;                  // User-facing message
  technical?: string;               // Technical details
  recoverable: boolean;             // Can user retry?
  suggestions?: string[];           // Suggested actions
  retryable: boolean;               // Can be retried
}

Skill

Skills are rendering capabilities with defined templates, dependencies, and execution parameters. 

interface SkillDefinition {
  // Identity
  id: string;                     // Skill ID (threejs, p5js, etc.)
  name: string;                   // Display name
  version: string;                // Skill version
  description: string;
  
  // Capabilities
  capabilities: Capability[];
  
  // Runtime
  runtime: RuntimeRequirements;
  
  // Templates
  templates: TemplateConfig;
  
  // Examples
  examples: SkillExample[];
  
  // Metadata
  author?: string;
  license?: string;
}

interface Capability {
  name: string;                   // Capability ID
  description: string;
  parameters: ParameterSchema;      // JSON Schema
}

interface RuntimeRequirements {
  baseImage: string;              // Docker image
  dependencies: string[];         // NPM packages
  envVars: Record;
  resources: ResourceLimits;
  ports: number[];
  filesystem: FilesystemConfig;
}

interface ResourceLimits {
  memory: string;                 // e.g., "512MB"
  cpu: string;                    // e.g., "1"
  timeout: number;                // Milliseconds
}

interface TemplateConfig {
  entryPoint: string;             // Main file
  files: string[];                // Template files
  schema?: string;                // Validation schema
}

interface SkillExample {
  prompt: string;                 // Example query
  code: string;                   // Example output
}

Task Plan

Task plans represent the orchestration pipeline structure with dependencies and execution status. 

interface TaskPlan {
  id: string;                     // Plan UUID
  turnId: string;                 // Associated turn
  
  // Tasks
  tasks: Task[];
  
  // Status
  status: "pending" | "running" | "completed" | "failed";
  
  // Progress
  progress: number;               // 0.0 to 1.0
  
  // Timing
  startedAt?: ISOString;
  completedAt?: ISOString;
  
  // Results
  results?: Record;
}

interface Task {
  id: string;                     // Task UUID
  action: TaskAction;             // Action type
  
  // Dependencies
  dependencies: string[];         // Task IDs to complete first
  
  // Status
  status: TaskStatus;
  
  // Progress
  progress: number;
  
  // Timing
  startedAt?: ISOString;
  completedAt?: ISOString;
  
  // Result
  result?: TaskResult;
  error?: TaskError;
}

type TaskAction = 
  | "parse_intent"
  | "select_skill"
  | "build_prompt"
  | "generate_code"
  | "validate_code"
  | "execute_code"
  | "sync_state";

type TaskStatus = "pending" | "running" | "completed" | "failed" | "skipped";

interface TaskResult {
  success: boolean;
  data?: any;
  metrics?: {
    duration: number;
    [key: string]: any;
  };
}

interface TaskError {
  code: string;
  message: string;
  recoverable: boolean;
}

Intent

Parsed user intent with entities, constraints, and confidence. 

interface ParsedIntent {
  // Original query
  rawQuery: string;
  normalizedQuery: string;
  
  // Classification
  intentType: IntentType;
  targetDomain: Domain;
  
  // Entities
  entities: Entity[];
  
  // Constraints
  constraints: Constraint[];
  
  // Confidence
  confidence: number;             // 0.0 to 1.0
  
  // Intent classification
  isGreeting: boolean;
  isCapabilityQuestion: boolean;
  isSmallTalk: boolean;
  isConversationOnly: boolean;
}

type IntentType = 
  | "create" 
  | "modify" 
  | "explain" 
  | "animate" 
  | "query" 
  | "chat";

type Domain = 
  | "3d" 
  | "2d" 
  | "data-viz" 
  | "diagram" 
  | "animation" 
  | "unknown";

interface Entity {
  type: string;                     // "object", "color", "animation", etc.
  value: string;
  normalized?: string;
  confidence: number;
  position?: { start: number; end: number };
}

interface Constraint {
  type: string;                     // "performance", "interactivity", "style"
  value: any;
  priority: "required" | "preferred" | "optional";
}

Error Types

enum ErrorType {
  // Intent parsing
  AMBIGUOUS_INTENT = "AMBIGUOUS_INTENT",
  UNSUPPORTED_DOMAIN = "UNSUPPORTED_DOMAIN",
  
  // Skill errors
  SKILL_NOT_FOUND = "SKILL_NOT_FOUND",
  SKILL_LOAD_FAILED = "SKILL_LOAD_FAILED",
  SKILL_INCOMPATIBLE = "SKILL_INCOMPATIBLE",
  
  // Code generation
  GENERATION_FAILED = "GENERATION_FAILED",
  VALIDATION_FAILED = "VALIDATION_FAILED",
  
  // Execution
  SYNTAX_ERROR = "SYNTAX_ERROR",
  RUNTIME_ERROR = "RUNTIME_ERROR",
  TIMEOUT_ERROR = "TIMEOUT_ERROR",
  RESOURCE_EXHAUSTED = "RESOURCE_EXHAUSTED",
  
  // System
  SANDBOX_UNAVAILABLE = "SANDBOX_UNAVAILABLE",
  NETWORK_ERROR = "NETWORK_ERROR",
  RATE_LIMITED = "RATE_LIMITED"
}

interface ErrorRecoveryStrategy {
  type: ErrorType;
  recoverable: boolean;
  actions: RecoveryAction[];
  fallback: FallbackBehavior;
}

type RecoveryAction = 
  | { type: "CLARIFY"; message: string }
  | { type: "SUGGEST"; options: string[] }
  | { type: "RETRY"; maxAttempts: number }
  | { type: "SIMPLIFY_PROMPT"; reduceComplexity: boolean }
  | { type: "USE_EXAMPLE"; fallbackToExample: boolean }
  | { type: "FIX_CODE"; useLLM: boolean }
  | { type: "ISOLATE_ERROR"; runPartial: boolean }
  | { type: "WAIT"; timeout: number }
  | { type: "SCALE_UP"; requestMoreSandboxes: boolean };

type FallbackBehavior =
  | { type: "ASK_USER" }
  | { type: "USE_DEFAULT"; skill: string }
  | { type: "SHOW_ERROR"; message: string }
  | { type: "SHOW_PARTIAL"; workingCode: boolean }
  | { type: "QUEUE"; notifyWhenReady: boolean };
